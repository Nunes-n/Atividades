    private void formWindowOpened(java.awt.event.WindowEvent evt) {                                  
       prepCon = new PreparaConexao("SYSTEM","537500");       
       prepCon.setDriver("oracle.jdbc.driver.OracleDriver");
       prepCon.setConnectionString("jdbc:oracle:thin:@127.0.0.1:1521:xe"); //"jdbc:oracle:thin:@192.168.1.6:1521:xe"
       daoHotel = new DaoHotel(prepCon.abrirConexao());
    }                                 

    private void formWindowClosed(java.awt.event.WindowEvent evt) {                                  
        prepCon.fecharConexao();
    }                                 

    private void btnSairActionPerformed(java.awt.event.ActionEvent evt) {                                        
        dispose();
    }                                       

    private void btnConsultarActionPerformed(java.awt.event.ActionEvent evt) {                                             
        hotel = null;
        hotel = daoHotel.consultar(Integer.parseInt(txtCodigo.getText()));
        
        if (hotel == null){//não encontrou o objeto na Base de Dados
           txtCodigo.setEnabled(false);
           txtNome.setEnabled(true);
           txtEndereco.setEnabled(true);
           txtTelefone.setEnabled(true);
           txtValorDiaria.setEnabled(true);
           txtNome.requestFocus();
           
           btnConsultar.setEnabled(false);
           btnIncluir.setEnabled(true);
           btnAlterar.setEnabled(false);
           btnExcluir.setEnabled(false);
       }
       else{ //encontrou o objeto na Base de Dados
          txtNome.setText(hotel.getNome());
          txtEndereco.setText(hotel.getEndereco());
          txtTelefone.setText(hotel.getTelefone());
          txtValorDiaria.setText(String.valueOf(hotel.getValorDiaria()));
          lblTotalFaturado.setText(String.valueOf(hotel.getTotalFaturamento()));
          txtCodigo.setEnabled(false); 
          txtNome.setEnabled(true);
          txtEndereco.setEnabled(true);
          txtTelefone.setEnabled(true);
          txtValorDiaria.setEnabled(true);
          txtNome.requestFocus();  
          
          btnConsultar.setEnabled(false);
          btnIncluir.setEnabled(false);
          btnAlterar.setEnabled(true);
          btnExcluir.setEnabled(true);
       }
    }                                            

    private void btnIncluirActionPerformed(java.awt.event.ActionEvent evt) {                                           
       hotel = new Hotel(Integer.parseInt(txtCodigo.getText()), 
                        txtNome.getText());
       hotel.setEndereco(txtEndereco.getText());
       hotel.setTelefone(txtTelefone.getText());
       hotel.setValorDiaria(Double.parseDouble(txtValorDiaria.getText()));
              
       //Realizando a inserção do objeto da classe Aluno
       //no SGBD via objeto da classe DAOAluno 
       daoHotel.inserir(hotel);
         
       //Incializando a GUI 
       txtCodigo.setText(null);
       txtNome.setText(null); 
       txtEndereco.setText(null);
       txtTelefone.setText(null);
       txtValorDiaria.setText(null);
       btnIncluir.setEnabled(false);
       txtCodigo.setEnabled(true);
       txtNome.setEnabled(false);
       txtEndereco.setEnabled(false);
       txtTelefone.setEnabled(false);
       txtValorDiaria.setEnabled(false);
       txtCodigo.requestFocus();        
       btnConsultar.setEnabled(true);
    }                                          

    private void btnAlterarActionPerformed(java.awt.event.ActionEvent evt) {                                           
        if (JOptionPane.showConfirmDialog(null, "Confirma Alteração?")== 0){    //Sim
           hotel.setNome(txtNome.getText());
           hotel.setEndereco(txtEndereco.getText());
           hotel.setTelefone(txtTelefone.getText());
           hotel.setValorDiaria(Double.parseDouble(txtValorDiaria.getText()));
           daoHotel.alterar(hotel);
        }      
        //Incializando a GUI 
        txtCodigo.setText(null);
        txtNome.setText(null);
        txtEndereco.setText(null);
        txtTelefone.setText(null);
        txtValorDiaria.setText(null);
        lblTotalFaturado.setText(null);
        txtCodigo.setEnabled(true); 
        txtNome.setEnabled(false);
        txtEndereco.setEnabled(false);
        txtTelefone.setEnabled(false);
        txtValorDiaria.setEnabled(false);
        txtCodigo.requestFocus();
        btnConsultar.setEnabled(true);
        btnIncluir.setEnabled(false);
        btnAlterar.setEnabled(false);
        btnExcluir.setEnabled(false);
    }                                          

    private void btnExcluirActionPerformed(java.awt.event.ActionEvent evt) {                                           
        if (JOptionPane.showConfirmDialog(null, "Confirma Exclusão?") == 0){
            daoHotel.excluir(hotel); 
            
            txtCodigo.setText(null);
            txtNome.setText(null);
            txtEndereco.setText(null);
            txtTelefone.setText(null);
            txtValorDiaria.setText(null);
            lblTotalFaturado.setText(null);
            txtCodigo.setEnabled(true); 
            txtNome.setEnabled(false);
            txtEndereco.setEnabled(false);
            txtTelefone.setEnabled(false);
            txtValorDiaria.setEnabled(false);
            txtCodigo.requestFocus();
            btnConsultar.setEnabled(true);
            btnIncluir.setEnabled(false);
            btnAlterar.setEnabled(false);
            btnExcluir.setEnabled(false);
        }
    }
